          setAnalyzing(false);

      disabled={analyzing}

/**
 * QuickPassAnalysisPanel - Shows analysis results inline
 */
function QuickPassAnalysisPanel({ intelligence }) {
  const qp = intelligence;
  const totalWarnings =
    qp.ambiguity.flags.length +
    qp.or_conflicts.issues.length +
    qp.marks_effort.warnings.length +
    qp.duplicates.pairs.length;

  const criticalIssues =
    qp.ambiguity.flags.filter((f) => f.severity === "high").length +
    qp.or_conflicts.issues.length;

  return (
    <div className="analysis-results">
      {/* Safety Summary */}
      <div className={`safety-banner ${qp.overall_safe ? "safe" : "warning"}`}>
        {qp.overall_safe
          ? "✅ Paper appears safe for evaluation"
          : "⚠️ Critical issues found - review before finalizing"}
      </div>

      {/* Checks Summary */}
      <div className="checks-grid">
        <CheckBadge
          icon={qp.ambiguity.safe ? "✅" : "⚠️"}
          label="Ambiguity"
          count={qp.ambiguity.flags.length}
          safe={qp.ambiguity.safe}
        />
        <CheckBadge
          icon={qp.or_conflicts.safe ? "✅" : "⚠️"}
          label="OR Questions"
          count={qp.or_conflicts.issues.length}
          safe={qp.or_conflicts.safe}
        />
        <CheckBadge
          icon={qp.marks_effort.safe ? "✅" : "⚠️"}
          label="Marks vs Effort"
          count={qp.marks_effort.warnings.length}
          safe={qp.marks_effort.safe}
        />
        <CheckBadge
          icon={qp.duplicates.safe ? "✅" : "⚠️"}
          label="Duplicates"
          count={qp.duplicates.pairs.length}
          safe={qp.duplicates.safe}
        />
      </div>

      {/* Warnings Detail */}
      {totalWarnings > 0 && (
        <div className="warnings-list">
          <h4>
            ⚠️ Issues Found ({totalWarnings}
            {criticalIssues > 0 && `, ${criticalIssues} critical`})
          </h4>

          {/* Ambiguity warnings */}
          {qp.ambiguity.flags.map((flag, i) => (
            <WarningCard
              key={`amb-${i}`}
              severity={flag.severity}
              title="Ambiguity Detected"
              issue={flag.issue}
              suggestion={flag.suggestion}
              qid={flag.qid}
            />
          ))}

          {/* OR conflicts */}
          {qp.or_conflicts.issues.map((issue, i) => (
            <WarningCard
              key={`or-${i}`}
              severity="high"
              title="OR Question Imbalance"
              issue={issue.issue}
              suggestion={issue.recommendation}
              qid={issue.pair.join(" vs ")}
            />
          ))}

          {/* Marks-effort warnings */}
          {qp.marks_effort.warnings.map((warn, i) => (
            <WarningCard
              key={`effort-${i}`}
              severity="medium"
              title="Marks Allocation Issue"
              issue={warn.reason}
              suggestion={`Adjust from ${warn.allocated_marks} to ${warn.recommended_marks} marks`}
              qid={warn.qid}
            />
          ))}

          {/* Duplicates */}
          {qp.duplicates.pairs.map((dup, i) => (
            <WarningCard
              key={`dup-${i}`}
              severity="medium"
              title="Duplicate Content"
              issue={dup.explanation}
              suggestion="Consider removing one or modifying"
              qid={dup.pair.join(" & ")}
            />
          ))}
        </div>
      )}
    </div>
  );
}

function CheckBadge({ icon, label, count, safe }) {
  return (
    <div className={`check-badge ${safe ? "check-safe" : "check-warn"}`}>
      <span className="check-icon">{icon}</span>
      <span className="check-label">{label}</span>
      <span className="check-count">{count === 0 ? "No issues" : `${count} issue${count === 1 ? "" : "s"}`}</span>
    </div>
  );
}

function WarningCard({ severity, title, issue, suggestion, qid }) {
  return (
    <div className={`warning-card severity-${severity}`}>
      <div className="warning-header">
        <span className="severity-badge">{severity}</span>
        <h5>{title}</h5>
        <span className="qid-tag">{qid}</span>
      </div>
      <p className="warning-issue">{issue}</p>
      <p className="warning-suggestion">💡 {suggestion}</p>
    </div>
  );
}
